cmake_minimum_required(VERSION 3.15)
project(ZiboCopilot CXX)
set(CMAKE_CXX_STANDARD 17)

set(XPSDK "${CMAKE_SOURCE_DIR}/SDK" CACHE PATH "X-Plane SDK path")
add_library(ZiboCopilot MODULE src/main.cpp src/copilot.cpp src/zibo_controls.cpp)
target_include_directories(ZiboCopilot PRIVATE ${XPSDK}/CHeaders/XPLM src)
target_compile_definitions(ZiboCopilot PRIVATE XPLM200 XPLM210 XPLM300 XPLM301)
set_target_properties(ZiboCopilot PROPERTIES PREFIX "" SUFFIX ".xpl" OUTPUT_NAME "ZiboCopilot")

if(WIN32)
  target_compile_definitions(ZiboCopilot PRIVATE IBM=1)
  target_link_directories(ZiboCopilot PRIVATE ${XPSDK}/Libraries/Win)
  target_link_libraries(ZiboCopilot XPLM_64)
elseif(APPLE)
  target_compile_definitions(ZiboCopilot PRIVATE APL=1)
  set(CMAKE_OSX_ARCHITECTURES "x86_64;arm64")
  target_link_options(ZiboCopilot PRIVATE -F${XPSDK}/Libraries/Mac -framework XPLM)
else()
  target_compile_definitions(ZiboCopilot PRIVATE LIN=1)
  target_compile_options(ZiboCopilot PRIVATE -fvisibility=hidden)
endif()
